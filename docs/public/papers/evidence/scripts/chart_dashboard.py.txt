from pathlib import Path
import json
import fitz
from reportlab.graphics.shapes import Drawing, String, Rect
from reportlab.graphics.charts.lineplots import LinePlot
from reportlab.graphics import renderSVG
from reportlab.lib.colors import HexColor

root=Path(__file__).resolve().parents[2]/'05-证据与数据'/'小米看板归档'
data={r:json.loads((root/('series-'+r+'.json')).read_text()) for r in ['flash','pro']}
b=json.loads((root/'benchmarks.json').read_text())['benchmarks'][0]['results']
d=Drawing(1000,630)
d.add(Rect(0,0,1000,630,fillColor=HexColor('#ffffff'),strokeColor=None))
d.add(String(35,603,'Xiaomi MiMo-V2.6 public RL archive',fontName='Helvetica-Bold',fontSize=19))
d.add(String(35,582,'Final retained steps | task mix changed | retrieved 2026-10-11',fontSize=11,fillColor=HexColor('#555555')))
metrics=[('Training avg@n','dynsam/avg@n',1),('DeepSWE v1.1: mini-swe-agent avg@3',None,1),('Trained tokens per step (billion)','perf/total_num_tokens',1e9),('Reported step time (hours)','timing_s/step',3600)]
for i,(title,key,scale) in enumerate(metrics):
 x=65+(i%2)*485;y=80+(1-i//2)*250
 p=LinePlot();p.x=x;p.y=y;p.width=400;p.height=170
 series=[]
 for r in ['flash','pro']:
  points=sorted((int(k),v) for k,v in b[r].items()) if key is None else [(s,v/scale) for s,v in zip(data[r]['steps'],data[r]['series'][key]) if v is not None]
  series.append(points)
 p.data=series;p.lines[0].strokeColor=HexColor('#d47c28');p.lines[1].strokeColor=HexColor('#326bb9');p.lines.strokeWidth=1.6
 p.xValueAxis.valueMin=1;p.xValueAxis.valueMax=30;p.xValueAxis.valueSteps=[1,5,10,15,20,25,30]
 vals=[v for points in series for _,v in points];span=max(vals)-min(vals)
 p.yValueAxis.valueMin=max(0,min(vals)-span*.1);p.yValueAxis.valueMax=max(vals)+span*.1
 p.xValueAxis.labels.fontSize=8;p.yValueAxis.labels.fontSize=8
 d.add(p);d.add(String(x,y+193,title,fontName='Helvetica-Bold',fontSize=12));d.add(String(x+180,y-25,'Training step',fontSize=10))
d.add(String(65,36,'Flash',fontSize=11,fillColor=HexColor('#d47c28')));d.add(String(120,36,'Pro',fontSize=11,fillColor=HexColor('#326bb9')))
d.add(String(180,36,'Lines connect observed points; missing evals are not filled. Rollback attempts omitted.',fontSize=10))
svg=renderSVG.drawToString(d)
(root/'训练与评测.svg').write_text(svg,encoding='utf-8')
doc=fitz.open(stream=svg.encode(),filetype='svg');pdf=doc.convert_to_pdf();doc.close()
doc=fitz.open(stream=pdf,filetype='pdf');doc[0].get_pixmap(matrix=fitz.Matrix(1.5,1.5)).save(str(root/'训练与评测.png'));doc.close()
print('chart generated with ReportLab LinePlot; rendered through MuPDF')

def panel_chart(filename,title,panels):
 drawing=Drawing(1000,850)
 drawing.add(Rect(0,0,1000,850,fillColor=HexColor('#ffffff'),strokeColor=None))
 drawing.add(String(35,821,title,fontName='Helvetica-Bold',fontSize=18))
 drawing.add(String(35,799,'Final retained series | missing values preserved | source: Xiaomi public RL archive',fontSize=10))
 for i,(label,series) in enumerate(panels):
  is_benchmark=len(panels)==3
  x=65 if is_benchmark else 65+(i%2)*485
  y=80+(2-i)*235 if is_benchmark else 80+(2-i//2)*235
  plot=LinePlot();plot.x=x;plot.y=y;plot.width=850 if is_benchmark else 400;plot.height=160;plot.data=series
  plot.lines[0].strokeColor=HexColor('#d47c28');plot.lines[1].strokeColor=HexColor('#326bb9');plot.lines.strokeWidth=1.5
  plot.xValueAxis.valueMin=1;plot.xValueAxis.valueMax=30;plot.xValueAxis.valueSteps=[1,5,10,15,20,25,30]
  vals=[v for points in series for _,v in points];span=max(vals)-min(vals) or 1
  plot.yValueAxis.valueMin=min(vals)-span*.1;plot.yValueAxis.valueMax=max(vals)+span*.1
  plot.xValueAxis.labels.fontSize=8;plot.yValueAxis.labels.fontSize=8
  drawing.add(plot);drawing.add(String(x,y+183,label,fontName='Helvetica-Bold',fontSize=11));drawing.add(String(x+(400 if is_benchmark else 180),y-22,'Training step',fontSize=10))
 drawing.add(String(65,35,'Flash',fontSize=11,fillColor=HexColor('#d47c28')));drawing.add(String(120,35,'Pro',fontSize=11,fillColor=HexColor('#326bb9')))
 drawing.add(String(180,35,'Curves show association, not causality. Training task mix changed.',fontSize=10))
 content=renderSVG.drawToString(drawing);(root/(filename+'.svg')).write_text(content,encoding='utf8')
 source=fitz.open(stream=content.encode(),filetype='svg');pdfdata=source.convert_to_pdf();source.close()
 result=fitz.open(stream=pdfdata,filetype='pdf');result[0].get_pixmap(matrix=fitz.Matrix(1.5,1.5)).save(str(root/(filename+'.png')));result.close()

panels=[]
for label,key in [('Policy gradient loss','actor/pg_loss'),('Entropy metric','actor/entropy_loss'),('Gradient norm before clipping','actor/grad_norm'),('New-infer / trainer KL','train_infer_diff/new_infer/kl'),('Mean trained-trajectory reward','critic/rewards/mean'),('Mean total context length (tokens)','ctx_total_length/mean')]:
